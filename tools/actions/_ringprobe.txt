(async()=>{
const p=window.__arenaProbe, cam=p.stage.camera, W=innerWidth, H=innerHeight;
const V=p.root.position.constructor;
const raf=window.requestAnimationFrame.bind(window);
await new Promise(r=>raf(r)); window.requestAnimationFrame=()=>0;
const out=[];
p.stage.scene.traverse(o=>{
  if(o.name!=='identity-ring') return;
  let cls='?'; for(let q=o.parent;q;q=q.parent){ if(q.name&&q.name.indexOf('critter-')===0){cls=q.name.slice(8);break;} }
  let band=null; o.traverse(c=>{ if(c.material&&c.material.uniforms&&c.material.uniforms.uBandA) band=c; });
  if(!band) return;
  const ws=band.getWorldScale(new V());
  const half=band.geometry.parameters.width/2*ws.x;
  const u=band.material.uniforms;
  const bIn=u.uBandA.value.z*half, bOut=u.uBandA.value.w*half;
  const kIn=u.uBandB.value.x*half, kOut=u.uBandB.value.y*half;
  const c0=band.getWorldPosition(new V());
  const proj=(x,z)=>{const v=new V(x,c0.y,z).project(cam);return [Math.round((v.x*.5+.5)*W*10)/10, Math.round((-v.y*.5+.5)*H*10)/10];};
  const pts=[];
  for(let a=0;a<360;a+=5){
    const rd=a*Math.PI/180, cs=Math.cos(rd), sn=Math.sin(rd);
    const at=(r)=>proj(c0.x+r*cs, c0.z+r*sn);
    pts.push({a, band:[at(bIn+(bOut-bIn)*0.3), at((bIn+bOut)/2), at(bIn+(bOut-bIn)*0.7)], ink:[at(kIn+(kOut-kIn)*0.35), at((kIn+kOut)/2), at(kIn+(kOut-kIn)*0.65)], gnd:[at(kOut*1.25), at(kOut*1.45)]});
  }
  out.push({cls, wx:+c0.x.toFixed(2), wz:+c0.z.toFixed(2), half:+half.toFixed(3), pts});
});
const ems=(p.emitters||[]).map(e=>({k:e.kind,x:+e.x.toFixed(2),z:+e.z.toFixed(2)}));
return {rings:out, emitters:ems};
})()
